<ng-template #searchModal>
  <div role="document" class="search-container">
    <div class="modal-header">
      <h5 class="modal-title">{{'HEADER.PRODUCT_SERACH' | translate}}</h5>
      <button type="button" class="btn-close" (click)="modalRef.hide()" aria-label="Close"></button>
    </div>
    <div class="modal-body position-relative">
      <div class="input-group">
        <input
          type="search"
          class="form-control"
          placeholder="{{'HEADER.ENTER_YOUR_SEARCH_TERM' | translate}}"
          [value]="searchQuery"
          (input)="onQueryInput($event.target['value'])"
          (keyup.enter)="doSearch()"
          autofocus
          autocomplete="off"
        />
        <button class="btn btn-outline-secondary" type="button" (click)="doSearch()">
          <i class="fa fa-search"></i>
        </button>
      </div>
      <div *ngIf="typeaheadLoading" class="text-center py-2">
        <apt-spinner [large]="false"></apt-spinner>
      </div>
      <div class="typeahead-results list-group mt-1" *ngIf="showResults">
        <button
          *ngFor="let product of searchResults"
          type="button"
          class="list-group-item list-group-item-action px-3 py-2"
          (click)="selectProduct(product)">
          <div class="d-flex align-items-center">
            <img class="me-3 flex-shrink-0" [src]="product?.IconId | image: true: true: product?.Id" [alt]="product?.Name" width="40" height="40" />
            <div class="text-truncate">
              <div class="fw-semibold">{{product?.Name}}</div>
              <small class="text-muted d-block">{{product?.ProductCode}}</small>
            </div>
          </div>
        </button>
      </div>
    </div>
    <div class="modal-footer">
      <button type="button" class="btn btn-link" (click)="modalRef.hide()">{{'COMMON.CANCEL' | translate}}</button>
      <button type="button" class="btn btn-primary btn-raised" (click)="doSearch()" [disabled]="searchQuery?.length < 3">{{'HEADER.SUBMIT' | translate}}</button>
    </div>
  </div>
</ng-template>
