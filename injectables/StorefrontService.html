constructor(private storeFrontService: StorefrontService) {}

private storeFrontService: StorefrontService = this.injector.get(StorefrontService);
}

getSettingSync
getSettingSync(key: string, defaultValue: T)
Type parameters :
  • T

Gets a setting value synchronously from the current state. Settings can be of various types (string, boolean, number, object).

// Get a string setting
const signature = this.getSettingSync<string>('Signature', '');
// Get a boolean setting
const isSandbox = this.getSettingSync<boolean>('IsSandbox', false);
Parameters :
Name Type Optional Default value Description
key string No

The setting key to retrieve
defaultValue T No null as T

The default value to return if the setting is not available (defaults to null)
Example : 
// Get a string setting
const signature = this.getSettingSync<string>(&#39;Signature&#39;, &#39;&#39;);
// Get a boolean setting

Returns : T

The setting value cast to type T, or defaultValue if not available

getStorefront
getStorefront()

Retrieves the storefront record

Example:

import { StorefrontService, Storefront } from '@congacommerce/ecommerce';
import { Observable } from 'rxjs/Observable';

export class MyComponent implements OnInit{
storefront: Storefront;
storefront$: Observable<Storefront>;

constructor(private storefrontService: StorefrontService){}

ngOnInit(){
this.storefrontService.getStorefront().subscribe(s => this.storefront = s);
// or
this.storefront$ = this.storefrontService.getStorefront();
}
}
Returns : Observable<Storefront>

an Observable containing the storefront record
